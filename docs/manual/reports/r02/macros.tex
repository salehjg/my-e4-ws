% ---- Listings -----------------------------------------------------------------
\lstdefinestyle{cmake}{
  language={},
  morekeywords={if,elseif,else,endif,set,add_flag,file,foreach,endforeach,
    add_executable,target_compile_options,target_compile_definitions,
    target_link_libraries,walberla_generate_sources,add_subdirectory,list,
    function,endfunction,option,STRINGS,REGEX,MATCHES,STREQUAL,PRIVATE,APPEND,
    LIMIT_COUNT,SCRIPTS,SCRIPT_ARGS,AND,NOT},
  keywordstyle=\color{kwcolor}\bfseries,
  comment=[l]{\#},
  commentstyle=\color{commentcolor},
  morestring=[b]",
  stringstyle=\color{strcolor},
}
\lstnewenvironment{cmakecode}[1][]{\lstset{style=cmake,#1}}{}
\lstdefinestyle{diff}{
  language={},
  moredelim=[il][\color{psred!85!black}]{---},
  moredelim=[l][\color{psred!85!black}]{-\ },
  moredelim=[l][\color{psteal!85!black}]{+\ },
  moredelim=[l][\color{psgray}]{@@},
}
\lstnewenvironment{diffcode}[1][]{\lstset{style=diff,#1}}{}

% ---- Page breaks ----------------------------------------------------------------
%   \rneedspace{len}: start a new page unless at least len of vertical space is left
%   (the classic needspace trick). Used before captioned listings so that a listing
%   does not start with its caption and one or two lines at the bottom of a page.
\makeatletter
\newcommand{\rneedspace}[1]{\par\penalty-100\begingroup
  \setlength{\dimen@}{#1}\dimen@ii\pagegoal\advance\dimen@ii-\pagetotal
  \ifdim\dimen@>\dimen@ii\ifdim\dimen@ii>\z@\vfil\fi\penalty-10000\fi\endgroup}
\makeatother
